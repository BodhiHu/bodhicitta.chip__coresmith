cmake_minimum_required(VERSION 3.16)
project(m1sim CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_library(m1sim STATIC m1_sim.cpp)
target_include_directories(m1sim PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(run_operators run_operators.cpp)
target_link_libraries(run_operators PRIVATE m1sim)
enable_testing()
add_test(NAME operators COMMAND run_operators)
