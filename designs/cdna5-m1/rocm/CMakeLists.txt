cmake_minimum_required(VERSION 3.16)
project(m1rocm CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../sim ${CMAKE_CURRENT_BINARY_DIR}/m1sim)
add_library(rocm_shim STATIC src/hip_shim.cpp src/hsa_shim.cpp)
target_include_directories(rocm_shim PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(rocm_shim PUBLIC m1sim)

foreach(ex hip_saxpy hip_gemm hip_softmax hip_flash_attention hsa_demo)
  add_executable(${ex} examples/${ex}.cpp)
  target_include_directories(${ex} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(${ex} PRIVATE rocm_shim)
endforeach()
enable_testing()
foreach(ex hip_saxpy hip_gemm hip_softmax hip_flash_attention hsa_demo)
  add_test(NAME ${ex} COMMAND ${ex})
endforeach()
